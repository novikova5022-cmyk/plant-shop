  <script>

    const tg = window.Telegram.WebApp;

    tg.expand(); // Разворачиваем на весь экран


    // Ваш список растений

    const products = [

      { id: 1, name: "👑 Алоказия Cucullata", price: 1200, img: "https://via.placeholder.com/300?text=Cucullata" },

      { id: 2, name: "🐉 Алоказия Dragon Scale", price: 1500, img: "https://via.placeholder.com/300?text=Dragon+Scale" },

      { id: 3, name: "🦎 Алоказия Jacklyn", price: 1800, img: "https://via.placeholder.com/300?text=Jacklyn" },

      { id: 4, name: "🌿 Монстера Classic", price: 1000, img: "https://via.placeholder.com/300?text=Monstera" },

      { id: 5, name: "🍓 Каладиум Strawberry Star", price: 900, img: "https://via.placeholder.com/300?text=Strawberry+Star" },

      { id: 6, name: "🌙 Каладиум Moonlight", price: 850, img: "https://via.placeholder.com/300?text=Moonlight" }

    ];


    let cart = {};


    // Отрисовка каталога

    const catalog = document.getElementById('catalog');

    products.forEach(p => {

      const card = document.createElement('div');

      card.className = 'card';

      card.innerHTML = `

        <img src="${p.img}" alt="${p.name}">

        <div class="title">${p.name}</div>

        <div class="price">${p.price} ₽</div>

        <button class="btn" id="btn-${p.id}" onclick="toggleCart(${p.id})">В корзину</button>

      `;

      catalog.appendChild(card);

    });


    // Управление корзиной

    function toggleCart(id) {

      const btn = document.getElementById(`btn-${id}`);

      if (cart[id]) {

        delete cart[id];

        btn.textContent = 'В корзину';

        btn.classList.remove('added');

      } else {

        cart[id] = 1;

        btn.textContent = '✓ В корзине';

        btn.classList.add('added');

      }

      updateMainButton();

    }


    // Главная нижняя кнопка Telegram

    function updateMainButton() {

      const count = Object.keys(cart).length;

      if (count > 0) {

        let total = 0;

        Object.keys(cart).forEach(id => {

          const item = products.find(p => p.id == id);

          total += item.price;

        });

        tg.MainButton.setText(`Оформить заказ (${total} ₽)`);

        tg.MainButton.show();

      } else {

        tg.MainButton.hide();

      }

    }


    // Отправка заказа в бот

    tg.MainButton.onClick(() => {

      let orderItems = [];

      Object.keys(cart).forEach(id => {

        const item = products.find(p => p.id == id);

        orderItems.push(item.name);

      });

      

      const data = {

        order: orderItems.join(", ")

      };

      

      tg.sendData(JSON.stringify(data));

      tg.close();

    });
